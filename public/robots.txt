# ============================================
# robots.txt - crawler rules
# Site: https://blog.xinchengp.cn
# Last updated: 2026-11-28
#
# NOTE: this file is intentionally kept pure ASCII.
# Baidu's robots parser is known to be unstable when a
# robots.txt contains non-ASCII bytes, and the Baidu
# Webmaster robots check tool reports a format error
# instead of showing the parsed rules. Keep all comments
# in English here, never in Chinese.
# ============================================

# Allow every search engine crawler
User-agent: *

# Allow every page
Allow: /

# Block admin pages (privacy and security)
Disallow: /admin
Disallow: /api/

# Block drafts and unfinished pages
Disallow: /draft/
Disallow: /temp/

# Block the music config (keep playlists away from AI crawlers)
Disallow: /content/music.json

# Sitemap location
Sitemap: https://blog.xinchengp.cn/sitemap.xml

# Explicit allow for Baiduspider
#
# Verified on 2026-11-28 through the Cloudflare API:
#   is_robots_txt_managed = false
#   cf_robots_variant     = off
# so Cloudflare is NOT injecting a managed block right now.
# If that flag is ever turned back on, Cloudflare prepends
# "User-agent: Baiduspider / Disallow: /" to this file and
# that block wins, because it is parsed before the origin
# copy. Re-check the flag before blaming the rules below.
User-agent: Baiduspider
Allow: /
Disallow: /admin
Disallow: /api/

# Bing specific config
User-agent: bingbot
Allow: /
Disallow: /admin
Disallow: /api/
Crawl-delay: 1
